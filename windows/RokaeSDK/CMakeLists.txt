cmake_minimum_required(VERSION 3.12)
project(rokae_experiment_offline VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(experiment_core src/core.cpp src/logger.cpp)
target_include_directories(experiment_core PUBLIC include)
target_link_libraries(experiment_core PUBLIC Threads::Threads)
if(MSVC)
  target_compile_options(experiment_core PRIVATE /W4 /fp:strict /utf-8)
else()
  target_compile_options(experiment_core PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math)
endif()
add_executable(experiment_offline src/main.cpp)
target_link_libraries(experiment_offline PRIVATE experiment_core)
add_executable(offline_tests tests/tests.cpp)
target_link_libraries(offline_tests PRIVATE experiment_core)
enable_testing()
add_test(NAME offline_tests COMMAND offline_tests)
# Deliberately no ROKAE DLL, socket library, robot connection or hardware target.
